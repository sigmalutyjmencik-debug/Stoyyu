local CacheFile = "EpsteinHub_Auth.json"

local function saveAuth(key)
    if writefile then
        local data = { SavedKey = key, ExpireTime = os.time() + 86400 }
        pcall(function() writefile(CacheFile, HttpService:JSONEncode(data)) end)
    end
end

local function checkSavedAuth()
    if readfile and isfile and isfile(CacheFile) then
        local ok, result = pcall(function() return HttpService:JSONDecode(readfile(CacheFile)) end)
        if ok and result and result.SavedKey and result.ExpireTime then
            if ValidKeys[result.SavedKey] == true and os.time() < result.ExpireTime then
                return result
            else
                pcall(function() delfile(CacheFile) end)
            end
        end
    end
    return nil
end

    function A.getSilentTarget()
        local best, bd = nil, A.FOV_RADIUS
        local vp = Camera.ViewportSize
        local center = Vector2.new(vp.X/2, vp.Y/2)
        for _,p in ipairs(Players:GetPlayers()) do
            if p == LP then continue end
            if A.TEAM_CHECK and p.Team == LP.Team then continue end
            if A.whitelistEnabled and A.whitelistPlayers[p.Name] then continue end
            local char = p.Character
            if not char then continue end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if not hum or hum.Health <= 0 then continue end
            local partName = A.HIT_PART
            if partName == "Random" then
                local opts = {"Head","Torso","HumanoidRootPart"}
                partName = opts[math.random(1,#opts)]
            end
            local part = char:FindFirstChild(partName) or char:FindFirstChild("Head")
            if not part then continue end
            local pos, onS = Camera:WorldToViewportPoint(part.Position)
            if not onS then continue end
            local dist = (Vector2.new(pos.X, pos.Y) - center).Magnitude
            if dist < bd then bd = dist best = part end
        end
        return best
    end

    function A.CreateHitmarker(worldPosition)
        if not A.hitMarkerEnabled then return end
        if not Drawing then return end
        local markerId = tostring(math.random(1,100000))
        local markerData = {
            WorldPosition = worldPosition,
            Size = A.hitMarkerSize,
            Gap = A.hitMarkerGap,
            Opacity = 1,
            Lines = {}
        }
        for i = 1,4 do
            local line = Drawing.new("Line")
            line.Visible = false
            line.Color = A.hitMarkerColor
            line.Thickness = 2.5
            table.insert(markerData.Lines, line)
        end
        A.ActiveHitmarkers[markerId] = markerData
        task.spawn(function()
            local duration = math.max(0.05, A.hitMarkerDuration)
            local totalSteps = 20
            local stepTime = duration / totalSteps
            for i = totalSteps, 0, -1 do
                if not A.ActiveHitmarkers[markerId] then return end
                markerData.Opacity = i / totalSteps
                task.wait(stepTime)
            end
            for _,line in ipairs(markerData.Lines) do pcall(function() line:Remove() end) end
            A.ActiveHitmarkers[markerId] = nil
        end)
    end

    local TARGET_ARM_NAMES = {["RightArm"]=true,["LeftArm"]=true,["SleeveL"]=true,["SleeveR"]=true}

    local function ForceHandsStyle(part)
        if not part or not part.Parent then return end
        local pbr = part:FindFirstChildOfClass("SurfaceAppearance")
        if pbr then pbr:Destroy() end
        local sm = part:FindFirstChildOfClass("SpecialMesh")
        if sm and sm.TextureId ~= "" then sm.TextureId = "" end
        if part:IsA("MeshPart") and part.TextureID ~= "" then part.TextureID = "" end
        for _,c in ipairs(part:GetChildren()) do
            if c:IsA("Texture") or c:IsA("Decal") or c:IsA("Clothing") then
                c:Destroy()
            end
        end
        part.Color = A.customHandsColor
        part.Transparency = A.customHandsTransp
        part.Material = Enum.Material.Neon
        pcall(function() part.LocalTransparencyModifier = A.customHandsTransp end)
    end

    local function BindArm(part)
        if A.activeLockedArms[part] then return end
        A.activeLockedArms[part] = true
        ForceHandsStyle(part)
        local busy = false
        part.Changed:Connect(function(prop)
            if busy or not part.Parent or not A.customHandsEnabled then return end
            if prop == "Color" or prop == "Transparency" or prop == "Material" or prop == "LocalTransparencyModifier" then
                busy = true
                part.Color = A.customHandsColor
                part.Transparency = A.customHandsTransp
                part.Material = Enum.Material.Neon
                pcall(function() part.LocalTransparencyModifier = A.customHandsTransp end)
                busy = false
            end
        end)
    end

    function A.SetupCustomHands()
        for _,obj in ipairs(game:GetDescendants()) do
            if obj:IsA("BasePart") and TARGET_ARM_NAMES[obj.Name] then BindArm(obj) end
        end
    end

    local function HideCharParts(char)
        for _, p in ipairs(char:GetChildren()) do
            if p:IsA("BasePart") and p.Name ~= "HumanoidRootPart" then
                p.LocalTransparencyModifier = 1
                p.Transparency = 1
                p.CanCollide = false
            elseif p:IsA("Accessory") or p:IsA("Shirt") or p:IsA("Pants") or p:IsA("CharacterMesh") then
                p:Destroy()
            elseif p:IsA("Decal") or p:IsA("Texture") then
                p.Transparency = 1
            end
        end
    end

    local function MountSahur()
        local char = LP.Character
        if not char then return end
        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return end

        if A.sahurPart and A.sahurPart.Parent == char then return end

        local assetId = (A.sahurModel == "Sahur 2") and A.sahurModel2 or A.sahurModel1
        local ok, objects = pcall(function()
            return game:GetObjects("rbxassetid://" .. assetId)
        end)
        if not ok or not objects or #objects == 0 then return end

        local loaded = objects[1]
        local mesh = loaded:IsA("MeshPart") and loaded or loaded:FindFirstChildOfClass("MeshPart")
        if not mesh then return end

        HideCharParts(char)

        local body = mesh:Clone()
        body.Name = "AstraSahurPart"
        body.CFrame = root.CFrame
        body.CanCollide = false
        body.Massless = true
        body.Parent = char

        local weld = Instance.new("WeldConstraint")
        weld.Part0 = root
        weld.Part1 = body
        weld.Parent = body

        A.sahurPart = body

        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.AutoRotate = false end

        ClearSahurBindings()

        A.sahurHideConn = RS.RenderStepped:Connect(function()
            if not A.sahurEnabled then return end
            local c = LP.Character
            if not c or not c.Parent then return end
            for _, p in ipairs(c:GetChildren()) do
                if p:IsA("BasePart") and p.Name ~= "HumanoidRootPart" then
                    if p.LocalTransparencyModifier ~= 1 then p.LocalTransparencyModifier = 1 end
                    if p.Transparency ~= 1 then p.Transparency = 1 end
                    if p.CanCollide then p.CanCollide = false end
                elseif p:IsA("Accessory") or p:IsA("Shirt") or p:IsA("Pants") or p:IsA("CharacterMesh") then
                    p:Destroy()
                end
            end
        end)

        A.sahurShiftLockConn = RS.RenderStepped:Connect(function()
            if not A.sahurEnabled then return end
            local c = LP.Character
            if not c or not c.Parent then return end
            local r = c:FindFirstChild("HumanoidRootPart")
            local h = c:FindFirstChildOfClass("Humanoid")
            if not r or not h or h.Health <= 0 then return end
            local look = A.Camera.CFrame.LookVector
            local rot = math.atan2(-look.X, -look.Z)
            r.CFrame = CFrame.new(r.Position) * CFrame.Angles(0, rot, 0)
        end)

        A.sahurHeartbeatConn = RS.Heartbeat:Connect(function()
            if not A.sahurEnabled then return end
            local c = LP.Character
            if not c then return end
            local r = c:FindFirstChild("HumanoidRootPart")
            if not r then return end
            local body2 = c:FindFirstChild("AstraSahurPart")
            if not body2 then
                A.sahurPart = nil
                MountSahur()
                return
            end
            body2.CanCollide = false
            body2.Massless = true
        end)

        if A.sahurSpinEnabled then
            A.sahurSpinConn = RS.RenderStepped:Connect(function(dt)
                if not A.sahurEnabled or not A.sahurSpinEnabled then return end
                local body2 = A.sahurPart
                if body2 and body2.Parent then
                    local cf = body2.CFrame
                    local newCF = cf * CFrame.Angles(0, math.rad(A.sahurSpinSpeed * 60 * dt), 0)
                    body2.CFrame = CFrame.new(cf.Position) * newCF.Rotation
                end
            end)
        end
    end

    local function UnmountSahur()
        ClearSahurBindings()
        if A.sahurPart then
            A.sahurPart:Destroy()
            A.sahurPart = nil
        end
        local char = LP.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.AutoRotate = true end
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") and p.Name ~= "HumanoidRootPart" then
                p.Transparency = 0
                p.LocalTransparencyModifier = 0
            end
        end
    end

        local char = LP.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")

    function A.DoAirWalk()
        if A.airWalkBusy then return end
        local char = LP.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not root or not hum then return end
        A.airWalkBusy = true
        local targetY = root.Position.Y + A.airWalkHeight
        root.CFrame = CFrame.new(root.Position.X, targetY, root.Position.Z) * root.CFrame.Rotation
        local bp = Instance.new("BodyPosition")
        bp.MaxForce = Vector3.new(0, math.huge, 0)
        bp.Position = Vector3.new(0, targetY, 0)
        bp.P = 25000
        bp.Parent = root
        hum:ChangeState(Enum.HumanoidStateType.Running)
        task.wait(A.airWalkDuration)
        if bp then bp:Destroy() end
        root.Velocity = Vector3.zero
        hum:ChangeState(Enum.HumanoidStateType.GettingUp)
        A.airWalkBusy = false
    end

    local function SetupDraggable(btn)
        local dragging, dragInput, dragStart, startPos
        btn.InputBegan:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = i.Position
                startPos = btn.Position
                i.Changed:Connect(function()
                    if i.UserInputState == Enum.UserInputState.End then dragging = false end
                end)
            end
        end)
        btn.InputChanged:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
                dragInput = i
            end
        end)
        UIS.InputChanged:Connect(function(i)
            if i == dragInput and dragging then
                local d = i.Position - dragStart
                btn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
            end
        end)
    end
